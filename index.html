import math as m import random as r import time import turtle as t # إعداد الشاشة ونافذة الرسم s = t.Screen() s.setup(700, 700) s.bgcolor("black") s.tracer(2) p = t.Turtle() p.hideturtle() # دالة حساب إحداثيات شكل القلب الرياضية def heart(a, scale): x = 16 * (m.sin(a) ** 3) * scale y = ( 13 * m.cos(a) - 5 * m.cos(2 * a) - 2 * m.cos(3 * a) - m.cos(4 * a) ) * scale return x, y # رسم الطبقة الأولى من الجسيمات for i in range(10000): a = r.uniform(0, 2 * m.pi) sc = r.uniform(6.5, 15.5) X, Y = heart(a, sc) ang = m.atan2(Y, X) r.uniform(-5.5, 0.5) length = r.uniform(4, 14) p.pencolor(0, r.uniform(0.25, 0.55), r.uniform(0.05, 0.85)) p.width(r.uniform(0.5, 1.2)) p.penup() p.goto(X, Y) p.pendown() p.goto(X + length * m.cos(ang), Y + length * m.sin(ang)) if i % 200 == 0: s.update() time.sleep(0.002) # رسم الطبقة الثانية والتفاصيل المتوهجة for i in range(15000): a = r.uniform(0, 2 * m.pi) X, Y = heart(a, 16.0) ang = m.atan2(Y, X) r.uniform(-0.5, 0.35) length = r.uniform(10, 22) p.pencolor(0, r.uniform(0.45, 0.75), r.uniform(0.75, 0.95)) p.width(r.uniform(0.4, 0.9)) p.penup() p.goto(X + r.uniform(-7, 2), r.uniform(2, 3)) p.pendown() p.goto(X + length * m.cos(ang), Y + length * m.sin(ang)) if i % 150 == 0: s.update() time.sleep(0.002) s.update() t.done()
